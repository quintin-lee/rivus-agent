# Rivus Agent Frontend Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a React + Vite + shadcn/ui single-page Dashboard frontend for the Rivus Agent Go backend, covering session management, task submission, real-time event streaming (polling-based), approval operations, and run history.

**Architecture:** Single-page layout with a 240px left sidebar (sessions + run history) and main area (RunForm + RunDetail + EventStream). Event streaming uses 2-second polling against the existing `GET /runs/{id}/events?after=N` endpoint. Zustand stores manage session, run, and UI state independently.

**Tech Stack:** React 18, Vite 7, TypeScript, Tailwind CSS 3, shadcn/ui, Zustand, Vitest + React Testing Library, pnpm

**Spec:** `docs/superpowers/specs/2026-10-09-frontend-design.md`

```ts
// src/lib/api.ts
import { API_BASE, OWNER_ID } from '@/App.config'
import type {
  AgentEvent, CreateRunReq, Run, RunStatus, Session, Step,
} from './types'

export class ApiError extends Error {
  code: string
  retryable: boolean
  constructor(code: string, message: string, retryable: boolean) {
    super(message)
    this.code = code
    this.retryable = retryable
  }
}

interface ApiResponse<T> {
  data: T
}

async function request<T>(
  method: string,
  path: string,
  body?: unknown,
  headers: Record<string, string> = {},
): Promise<T> {
  const res = await fetch(`${API_BASE}${path}`, {
    method,
    headers: {
      'Content-Type': 'application/json',
      'X-Owner-ID': OWNER_ID,
      ...headers,
    },
    body: body ? JSON.stringify(body) : undefined,
  })
  if (!res.ok) {
    let errBody: { code?: string; message?: string; retryable?: boolean } = {}
    try { errBody = await res.json() } catch { /* ignore */ }
    throw new ApiError(
      errBody.code ?? String(res.status),
      errBody.message ?? res.statusText,
      errBody.retryable ?? false,
    )
  }
  if (res.status === 204) return undefined as T
  return res.json() as Promise<T>
}

export class ApiClient {
  async createSession(title: string): Promise<{ session_id: string }> {
    return request('POST', '/api/v1/sessions', { title })
  }

  async getSession(id: string): Promise<Session> {
    return request('GET', `/api/v1/sessions/${id}`)
  }

  async listSessions(): Promise<Session[]> {
    // Backend endpoint may not exist yet; catch and return []
    try {
      const res = await request<{ sessions: Session[] }>('GET', '/api/v1/sessions')
      return res.sessions ?? []
    } catch {
      return []
    }
  }

  async listRuns(sessionId: string, limit = 50): Promise<Run[]> {
    try {
      const res = await request<{ runs: Run[] }>(
        'GET', `/api/v1/runs?session_id=${encodeURIComponent(sessionId)}&limit=${limit}`,
      )
      return res.runs ?? []
    } catch {
      return []
    }
  }

  async createRun(
    req: CreateRunReq,
    idempotencyKey?: string,
  ): Promise<{ run_id: string; status: RunStatus; duplicated: boolean }> {
    return request(
      'POST', '/api/v1/runs', req,
      idempotencyKey ? { 'Idempotency-Key': idempotencyKey } : {},
    )
  }

  async getRun(id: string): Promise<{ run: Run; steps: Step[] }> {
    return request('GET', `/api/v1/runs/${id}`)
  }

  async getEvents(runId: string, afterSeq: number): Promise<AgentEvent[]> {
    // Backend returns a JSON array directly via SSE text or the existing one-shot handler.
    // For polling, call the one-shot handler which returns all events after `afterSeq`.
    const res = await fetch(
      `${API_BASE}/api/v1/runs/${runId}/events?after=${afterSeq}`,
      { headers: { 'X-Owner-ID': OWNER_ID } },
    )
    if (!res.ok) {
      let errBody: { code?: string; message?: string } = {}
      try { errBody = await res.json() } catch { /* ignore */ }
      throw new ApiError(errBody.code ?? String(res.status), errBody.message ?? 'fetch events failed', false)
    }
    // The handler writes SSE text frames; parse them manually
    const text = await res.text()
    const events: AgentEvent[] = []
    for (const block of text.split('\n\n')) {
      if (!block.trim()) continue
      let id = '', eventType = '', data = ''
      for (const line of block.split('\n')) {
        if (line.startsWith('id: ')) id = line.slice(4)
        else if (line.startsWith('event: ')) eventType = line.slice(7)
        else if (line.startsWith('data: ')) data = line.slice(6)
      }
      if (eventType && data) {
        events.push({
          seq: parseInt(id, 10) || 0,
          run_id: runId,
          event_type: eventType as AgentEvent['event_type'],
          payload_json: data,
          sensitivity: 'normal',
          created_at: 0,
        })
      }
    }
    return events
  }

  async cancelRun(id: string): Promise<void> {
    await request('POST', `/api/v1/runs/${id}/cancel`)
  }

  async resumeRun(id: string): Promise<void> {
    await request('POST', `/api/v1/runs/${id}/resume`)
  }

  async decideApproval(
    runId: string, approvalId: string,
    approve: boolean, reason?: string,
  ): Promise<void> {
    await request('POST', `/api/v1/runs/${runId}/approvals/${approvalId}`, {
      approve, reason,
    })
  }
}

export const api = new ApiClient()
```

```ts
// src/lib/poller.ts
import type { AgentEvent, RunStatus } from './types'

export type PollStatus = 'polling' | 'error' | 'stopped'

export interface PollerOpts {
  getEvents: (runId: string, afterSeq: number) => Promise<AgentEvent[]>
  getRunStatus: (runId: string) => Promise<RunStatus>
  runId: string
  intervalMs?: number
  getAfterSeq: () => number
  onEvent: (e: AgentEvent) => void
  onStatus: (s: PollStatus) => void
}

const TERMINAL: Set<RunStatus> = new Set([
  'succeeded', 'failed', 'cancelled', 'timed_out',
])

export function startEventPoller(opts: PollerOpts): () => void {
  const interval = opts.intervalMs ?? 2000
  let afterSeq = opts.getAfterSeq()
  let stopped = false
  let timer: ReturnType<typeof setTimeout> = null!
  let backoffMs = 0
  let pollTimer: ReturnType<typeof setInterval> | null = null

  async function tick() {
    if (stopped) return
    try {
      const events = await opts.getEvents(opts.runId, afterSeq)
      backoffMs = 0
      opts.onStatus('polling')
      for (const e of events) {
        opts.onEvent(e)
        if (e.seq > afterSeq) afterSeq = e.seq
      }
      // Check if run reached terminal state
      const status = await opts.getRunStatus(opts.runId)
      if (TERMINAL.has(status)) {
        stop()
        opts.onStatus('stopped')
        return
      }
    } catch {
      if (!stopped) {
        opts.onStatus('error')
        backoffMs = Math.min(backoffMs + 1000, 30_000)
        timer = setTimeout(tick, backoffMs)
        return
      }
    }
    if (!stopped) {
      timer = setTimeout(tick, interval)
    }
  }

  function stop() {
    stopped = true
    clearTimeout(timer)
    if (pollTimer) clearInterval(pollTimer)
  }

  timer = setTimeout(tick, 0)
  pollTimer = setInterval(tick, interval) // keep-alive reference
  return stop
}
```

```ts
import { create } from 'zustand'
import { api } from '@/lib/api'
import type { Session } from '@/lib/types'

interface SessionState {
  sessions: Session[]
  activeSessionId: string | null
  createSession: (title: string) => Promise<void>
  switchSession: (id: string) => void
  loadSessions: () => Promise<void>
}

export const useSessionStore = create<SessionState>((set, get) => ({
  sessions: [],
  activeSessionId: null,

  createSession: async (title: string) => {
    const { session_id } = await api.createSession(title)
    set((s) => ({
      sessions: [...s.sessions, { session_id, title, created_at: Date.now(), updated_at: Date.now() }],
      activeSessionId: session_id,
    }))
  },

  switchSession: (id: string) => set({ activeSessionId: id }),

  loadSessions: async () => {
    const sessions = await api.listSessions()
    set({ sessions })
    if (sessions.length > 0 && !get().activeSessionId) {
      set({ activeSessionId: sessions[0].session_id })
    }
  },
}))
```

```ts
// src/store/run-store.ts
import { create } from 'zustand'
import { api } from '@/lib/api'
import { MAX_EVENTS } from '@/App.config'
import type { AgentEvent, EventType, Run, Step } from '@/lib/types'

type EventFilter = EventType | 'all'

interface RunState {
  activeRunId: string | null
  activeRun: Run | null
  activeSteps: Step[]
  events: AgentEvent[]
  eventFilter: EventFilter
  runHistory: Run[]

  activateRun: (id: string) => Promise<void>
  appendEvent: (e: AgentEvent) => void
  setEventFilter: (f: EventFilter) => void
  clearEvents: () => void
  loadRunHistory: (sessionId: string) => Promise<void>
  fetchRunAndSteps: (id: string) => Promise<void>
}

export const useRunStore = create<RunState>((set, get) => ({
  activeRunId: null,
  activeRun: null,
  activeSteps: [],
  events: [],
  eventFilter: 'all',
  runHistory: [],

  activateRun: async (id: string) => {
    set({ activeRunId: id, events: [], eventFilter: 'all' })
    await get().fetchRunAndSteps(id)
  },

  fetchRunAndSteps: async (id: string) => {
    try {
      const { run, steps } = await api.getRun(id)
      set({ activeRun: run, activeSteps: steps })
    } catch (err) {
      if (err instanceof Error && err.message.includes('404')) {
        // Run not found — leave activeRun as null
      }
    }
  },

  appendEvent: (e: AgentEvent) => {
    set((s) => {
      const events = [...s.events, e]
      // Truncate to MAX_EVENTS (keep most recent)
      if (events.length > MAX_EVENTS) {
        return { ...s, events: events.slice(-MAX_EVENTS) }
      }
      return { ...s, events }
    })
    // Update activeRun status from terminal events
    if (e.event_type === 'run.finished' || e.event_type === 'run.failed' ||
        e.event_type === 'run.cancelled') {
      get().fetchRunAndSteps(e.run_id)
    }
  },

  setEventFilter: (eventFilter) => set({ eventFilter }),
  clearEvents: () => set({ events: [] }),

  loadRunHistory: async (sessionId: string) => {
    const runs = await api.listRuns(sessionId, 20)
    set({ runHistory: runs })
  },
}))
```

```tsx
export function ApprovalCard({ event }: ApprovalCardProps) {
  const runId = event.run_id
  const [rejectOpen, setRejectOpen] = useState(false)
  const [reason, setReason] = useState('')
  const [decided, setDecided] = useState<'approved' | 'rejected' | 'expired' | null>(null)
  const [toolName, setToolName] = useState('')
  const [approvalId, setApprovalId] = useState('')
  const [expiresAt, setExpiresAt] = useState(0)

  // Parse approval payload once on mount
  useEffect(() => {
    try {
      const p = JSON.parse(event.payload_json)
      setToolName(p.tool_name ?? p.tool ?? 'unknown')
      setApprovalId(p.approval_id ?? p.id ?? '')
      setExpiresAt(p.expires_at ?? 0)
    } catch { /* ignore malformed payload */ }
  }, [event.payload_json])

  const now = Date.now()
  const isExpired = decided === null && expiresAt > 0 && now > expiresAt

  useEffect(() => {
    if (!isExpired) return
    const interval = setInterval(() => {
      if (Date.now() > expiresAt && !decided) setDecided('expired')
    }, 1000)
    return () => clearInterval(interval)
  }, [isExpired, expiresAt, decided])

  const handleApprove = async () => {
    try {
      await api.decideApproval(runId, approvalId, true)
      setDecided('approved')
      useRunStore.getState().fetchRunAndSteps(runId)
    } catch (err) {
      console.error('approval failed', err)
    }
  }

  const handleReject = async () => {
    try {
      await api.decideApproval(runId, approvalId, false, reason)
      setDecided('rejected')
      setRejectOpen(false)
      useRunStore.getState().fetchRunAndSteps(runId)
    } catch (err) {
      console.error('reject failed', err)
    }
  }

  // Determine effective state: local decided or expired
  const effectiveState = decided ?? (isExpired ? 'expired' : 'pending')

  return (
    <div className="my-2 rounded-lg border border-zinc-700 bg-zinc-800/80 p-3">
      <div className="flex items-center justify-between">
        <div>
          <span className="text-sm font-medium text-zinc-100">{toolName}</span>

                    effectiveState === 'rejected' ? 'destructive' : 'secondary'}
```

```tsx
      case 'tool.completed':
      case 'tool.failed':
        return `tool: ${p.tool_name ?? p.tool ?? 'unknown'}`
      case 'model.requested':
        return `model call #${p.call_index ?? '?'}`
      case 'run.resumed':
        return `Run resumed (attempt ${p.attempt ?? '?'})`
      case 'approval.requested':
        return `Approval requested: ${p.tool_name ?? p.tool ?? 'unknown'}`
      case 'approval.decided':
        return p.approved ? 'Approved' : `Rejected: ${p.reason ?? ''}`
      case 'verification.passed':
        return 'Verification passed'
      case 'verification.failed':
        return `Verification failed: ${p.reason ?? ''}`
      default:
        return e.payload_json.length > 80 ? e.payload_json.slice(0, 80) + '…' : e.payload_json
    }
  } catch {
    return e.payload_json.length > 80 ? e.payload_json.slice(0, 80) + '…' : e.payload_json
  }
}

export function EventStream() {
  const events = useRunStore((s) => s.events)
  const eventFilter = useRunStore((s) => s.eventFilter)
  const setEventFilter = useRunStore((s) => s.setEventFilter)
  const eventStreamOpen = useUiStore((s) => s.eventStreamOpen)
  const toggleEventStream = useUiStore((s) => s.toggleEventStream)

  const [search, setSearch] = useState('')
  const scrollRef = useRef<HTMLDivElement>(null)
  const [autoScroll, setAutoScroll] = useState(true)
  const [showNewBadge, setShowNewBadge] = useState(false)

  const filtered = useMemo(() => {
    return events.filter((e) => {
      if (eventFilter !== 'all') {
        const cat = eventTypeCategory(e.event_type)
        if (eventFilter.startsWith('run') && cat !== 'run' && cat !== 'plan' && cat !== 'checkpoint') {
          // allow matching specific event types or category
          if (e.event_type !== eventFilter && !e.event_type.startsWith(eventFilter)) return false
        }
        if (eventFilter.startsWith('model') && cat !== 'model') {
          if (e.event_type !== eventFilter && !e.event_type.startsWith('model.')) return false
        }
        if (eventFilter.startsWith('tool') && cat !== 'tool') {
          if (e.event_type !== eventFilter && !e.event_type.startsWith('tool.')) return false
        }
        if (eventFilter.startsWith('step') && cat !== 'step') {
          if (e.event_type !== eventFilter && !e.event_type.startsWith('step.')) return false
        }
        if (eventFilter.startsWith('approval') && cat !== 'approval') {
          if (e.event_type !== eventFilter && !e.event_type.startsWith('approval.')) return false
        }
      }
      if (search && !e.payload_json.toLowerCase().includes(search.toLowerCase())) return false
      return true
    })
  }, [events, eventFilter, search])

  // Auto-scroll
  useEffect(() => {
    if (autoScroll && scrollRef.current) {
      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
    }
    if (events.length > 0) setShowNewBadge(true)
  }, [events, autoScroll])

  const handleScroll = useCallback(() => {
    if (!scrollRef.current) return
    const { scrollTop, scrollHeight, clientHeight } = scrollRef.current
    if (scrollHeight - scrollTop - clientHeight > 50) {
      setAutoScroll(false)
    } else {
      setAutoScroll(true)
      setShowNewBadge(false)
    }
  }, [])

  const scrollToBottom = () => {
    if (scrollRef.current) {
      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
      setAutoScroll(true)
      setShowNewBadge(false)
    }
  }

  if (!eventStreamOpen) return null

  return (
    <div className="flex flex-col rounded-lg border border-zinc-800 bg-zinc-950">
      {/* Toolbar */}
      <div className="flex items-center gap-2 border-b border-zinc-800 px-3 py-2">
        <span className="text-xs font-medium text-zinc-400">Event Stream</span>
        <div className="flex gap-1">
          {FILTERS.map((f) => (
            <button
              key={f.value}
              className={cn(
                'rounded px-1.5 py-0.5 text-[10px] transition-colors',
                eventFilter === f.value
                  ? 'bg-zinc-700 text-zinc-100'
                  : 'text-zinc-500 hover:text-zinc-300',
              )}
              onClick={() => setEventFilter(f.value as EventType | 'all')}
            >
              {f.label}
            </button>
          ))}
        </div>
        <div className="flex-1" />
        <div className="relative">
          <Search className="absolute left-1.5 top-1/2 h-3 w-3 -translate-y-1/2 text-zinc-600" />
          <Input
            placeholder="Search…"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="h-6 w-40 pl-6 text-xs bg-zinc-900 border-zinc-700"
          />
        </div>
        <button
          className="text-xs text-zinc-600 hover:text-zinc-300"
          onClick={toggleEventStream}
        >
          ✕
        </button>
      </div>

      {/* Event list */}
      <div
        ref={scrollRef}
        onScroll={handleScroll}
        className="relative max-h-96 overflow-y-auto p-3 font-mono text-xs"
      >
        {filtered.length === 0 && events.length === 0 && (
          <p className="text-zinc-600">No events yet…</p>
        )}
        {filtered.length === 0 && events.length > 0 && (
          <p className="text-zinc-600">No matching events.</p>
        )}

        {filtered.map((e) => (
          <div key={e.seq}>
            <div
              className={cn(
                'flex items-start gap-2 py-0.5',
                e.sensitivity === 'high' && 'opacity-60',
              )}
            >
              <span className="shrink-0 text-zinc-600">{formatTimestamp(e.created_at)}</span>
              <Badge
                variant="outline"
                className={cn(
                  'shrink-0 border-0 text-[10px]',
                  EVENT_COLORS[e.event_type] ?? 'bg-zinc-700 text-zinc-400',
                )}
              >
                {e.event_type}
              </Badge>
              <span className="text-zinc-400">
                {e.sensitivity === 'high' ? 'masked' : eventSummary(e)}
              </span>
            </div>

            {e.event_type === 'approval.requested' && (
              <ApprovalCard event={e} />
            )}
          </div>
        ))}

        {/* New events badge */}
        {showNewBadge && !autoScroll && (
          <button
            onClick={scrollToBottom}
            className="absolute bottom-2 right-3 rounded-full bg-blue-600 px-2.5 py-0.5 text-[10px] text-white shadow-lg"
          >
            ↓ New
          </button>
        )}
      </div>
    </div>
  )
}
```

```go
func (r *TaskRepo) ListSessions(ctx context.Context, ownerID string) ([]struct {
    ID, Title string
    CreatedAt, UpdatedAt int64
}, error) {
    rows, err := r.db.QueryContext(ctx,
        `SELECT id, title, created_at, updated_at FROM sessions WHERE owner_id = ? ORDER BY updated_at DESC`,
        ownerID)
    if err != nil {
        return nil, err
    }
    defer func() { _ = rows.Close() }()
    var out []struct {
        ID, Title string
        CreatedAt, UpdatedAt int64
    }
    for rows.Next() {
        var s struct {
            ID, Title string
            CreatedAt, UpdatedAt int64
        }
        var title sql.NullString
        if err := rows.Scan(&s.ID, &title, &s.CreatedAt, &s.UpdatedAt); err != nil {
            return nil, err
        }
        s.Title = strOr(title, "")
        out = append(out, s)
    }
    return out, rows.Err()
}
```

```go
func (s *Server) handleListSessions(w http.ResponseWriter, r *http.Request) {
    sessions, err := s.tasks.ListSessions(r.Context(), ownerOf(r))
    if err != nil {
        writeErr(w, 500, "internal", "list sessions failed", true)
        return
    }
    writeJSON(w, 200, map[string]any{"sessions": sessions})
}

// Handler: list runs (filter by session_id query param)
func (s *Server) handleListRuns(w http.ResponseWriter, r *http.Request) {
    owner := ownerOf(r)
    sessionID := r.URL.Query().Get("session_id")
    if sessionID == "" {
        writeErr(w, 400, "bad_request", "session_id query param required", false)
        return
    }
    limit := 50
    if v := r.URL.Query().Get("limit"); v != "" {
        if n, err := strconv.Atoi(v); err == nil && n > 0 && n <= 100 {
            limit = n
        }
    }
    runs, err := s.tasks.ListRunsBySession(r.Context(), owner, sessionID, limit)
    if err != nil {
        writeErr(w, 404, "not_found", "session not found", false)
        return
    }
    writeJSON(w, 200, map[string]any{"runs": runs})
}
```
